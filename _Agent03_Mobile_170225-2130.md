### **3.2 Mobile Dashboard Tests (15 Min)**

```typescript
test('Mobile Dashboard loads correctly', async ({ page }) => {
  await page.goto('http://localhost:3000/m');

  // Check auth redirect (if not logged in)
  await expect(page).toHaveURL(/\/login-pin|\/m/);

  // If on dashboard:
  if (page.url().includes('/m')) {
    // Check header
    await expect(page.locator('h1')).toContainText('GreekLingua');

    // Check 12 module tiles
    const tiles = page.locator('[data-testid="module-tile"]');
    await expect(tiles).toHaveCount(12);

    // Check bottom nav
    const bottomNav = page.locator('[data-testid="bottom-nav"]');
    await expect(bottomNav).toBeVisible();
  }
});

test('Mobile Bottom Navigation works', async ({ page }) => {
  await page.goto('http://localhost:3000/m');

  // Click Stats
  await page.click('text=Stats');
  await expect(page).toHaveURL('/m/stats');

  // Click Settings
  await page.click('text=Settings');
  await expect(page).toHaveURL('/m/settings');

  // Click Home
  await page.click('text=Home');
  await expect(page).toHaveURL('/m');
});
```

### **3.3 Practice Modes Mobile Tests (30 Min)**

```typescript
test('Practice Modes Mobile UI loads', async ({ page }) => {
  await page.goto('http://localhost:3000/m/practice-modes');

  // Check header
  await expect(page.locator('h1')).toContainText('Practice Modes');

  // Check practice items list
  const items = page.locator('[data-testid="practice-item"]');
  await expect(items.first()).toBeVisible();
});

test('Practice Mode Bottom Sheet opens', async ({ page }) => {
  await page.goto('http://localhost:3000/m/practice-modes');

  // Click first practice item
  const firstItem = page.locator('[data-testid="practice-item"]').first();
  await firstItem.click();

  // Check bottom sheet opened
  const bottomSheet = page.locator('[data-testid="practice-modes-sheet"]');
  await expect(bottomSheet).toBeVisible();

  // Check game mode buttons
  await expect(page.locator('text=Matching Game')).toBeVisible();
  await expect(page.locator('text=Multiple Choice')).toBeVisible();
  await expect(page.locator('text=Write It Out')).toBeVisible();
});

test('Matching Game works on mobile', async ({ page }) => {
  await page.goto('http://localhost:3000/m/practice-modes');

  // Open practice item
  await page.click('[data-testid="practice-item"]');

  // Click Matching Game
  await page.click('text=Matching Game');

  // Wait for game to load
  await page.waitForSelector('[data-testid="matching-card"]');

  // Click 2 cards (simulate match)
  const cards = page.locator('[data-testid="matching-card"]');
  await cards.nth(0).click();
  await cards.nth(1).click();

  // Check if match detected (score updated)
  const score = page.locator('[data-testid="score"]');
  await expect(score).toBeVisible();
});
```

### **3.4 Vocabulary Mobile Tests (30 Min)**

```typescript
test('Vocabulary Mobile UI loads', async ({ page }) => {
  await page.goto('http://localhost:3000/m/vocabulary');

  // Check header
  await expect(page.locator('h1')).toContainText('Vocabulary');

  // Check card visible
  const card = page.locator('[data-testid="vocab-card"]');
  await expect(card).toBeVisible();

  // Check progress indicator
  await expect(page.locator('text=/Card \\d+ \\/ \\d+/')).toBeVisible();
});

test('Vocab card flip works', async ({ page }) => {
  await page.goto('http://localhost:3000/m/vocabulary');

  // Card shows Greek (front)
  await expect(page.locator('text=Greek')).toBeVisible();

  // Tap card to flip
  await page.click('[data-testid="vocab-card"]');

  // Card shows English (back)
  await expect(page.locator('text=English')).toBeVisible();

  // Rating buttons visible
  await expect(page.locator('text=Again')).toBeVisible();
  await expect(page.locator('text=Easy')).toBeVisible();
});

test('Vocab rating buttons work', async ({ page }) => {
  await page.goto('http://localhost:3000/m/vocabulary');

  // Flip card
  await page.click('[data-testid="vocab-card"]');

  // Click "Good" button
  await page.click('text=Good');

  // Next card should load
  await expect(page.locator('[data-testid="vocab-card"]')).toBeVisible();

  // Progress should update (Card 2 / X)
  await expect(page.locator('text=/Card \\d+ \\/ \\d+/')).toBeVisible();
});
```

## 💡 TIPPS

**Playwright Mobile Testing:**
```typescript
// Mobile viewport
test.use(devices['iPhone 12']);

// Touch gestures
await page.tap('[data-testid="button"]');

// Swipe
await page.mouse.down();
await page.mouse.move(100, 0);
await page.mouse.up();
```

**Lighthouse Best Practices:**
- Run on production build (`npm run build && npm start`)
- Clear cache before test
- Use mobile network throttling (Fast 4G)
- Test multiple times (take average)

**Screen Reader Testing:**
- iOS: Settings → Accessibility → VoiceOver
- Android: Settings → Accessibility → TalkBack
- Navigate with swipe gestures
- Test all interactive elements
